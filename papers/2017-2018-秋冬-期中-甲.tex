\begin{problem}{15 分}{8.5cm}
计算 $n$ 阶行列式
\[
\begin{vmatrix}
x&y&y&\cdots&y\\
z&x&y&\cdots&y\\
z&z&x&\cdots&y\\
\vdots&\vdots&\vdots&&\vdots\\
z&z&z&\cdots&x
\end{vmatrix},
\]
$x,y,z$ 为任意实常数。
\end{problem}

\begin{problem}{20 分}{12cm}
设 $k$ 为实常数，当 $k$ 为何值时，下面线性方程组无解？唯一解？无穷多解？有解时，求解。
\[
\begin{cases}
kx_1+x_2+x_3=k-3,\\
x_1+kx_2+x_3=-2,\\
x_1+x_2+kx_3=-2.
\end{cases}
\]
\end{problem}

\begin{problem}{20 分}{10cm}
求矩阵方程
\[
X\begin{pmatrix}1&1&1\\0&1&2\\1&0&2\end{pmatrix}=\begin{pmatrix}1&1&1\\0&0&-6\end{pmatrix}.
\]
\end{problem}

\begin{problem}{15 分}{8.5cm}
设 $R(A_{m\times n})=r$，证明存在 $B_{n\times n}$，且 $R(B)=n-r$，使得 $AB=0$。
\end{problem}

\begin{problem}{15 分}{9.5cm}
设 $R(A_{n\times n})=1$，$A_{n\times n}\in P^{n\times n}$，证明：
\begin{enumerate}[label=(\arabic*),leftmargin=2.2em,itemsep=0.25em,topsep=0.3em]
  \item 存在两组不全为零的实数 $a_1,\cdots,a_n$；$b_1,\cdots,b_n$，使得 $A=(a_1,\cdots,a_n)^{T}(b_1,\cdots,b_n)$；
  \item 存在实数 $k$，使得 $A^{2}=kA$。
\end{enumerate}
\end{problem}

\begin{problem}{8 分}{6.5cm}
设 $A_{m\times n}X_{n\times 1}=d_{m\times 1}$ 有解，$B_{m\times s}X_{s\times 1}=c_{m\times 1}$ 无解，令 $G=(ABdc)_{m\times(n+s+2)}$，证明：$R(G)\le R(A)+R(B)+1$。
\end{problem}

\begin{problem}{10 分}{7cm}
设 $A,B,C,D\in R^{n\times n}$，证明：当 $AC=CA$ 时，有
$\begin{vmatrix}A&D\\C&B\end{vmatrix}=|AB-CD|$。
\end{problem}
