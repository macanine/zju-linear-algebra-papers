\begin{problem}{10 分}{7cm}
试计算行列式
\[
\begin{vmatrix}
1&0&2&0&x\\
0&2&0&1&0\\
1&0&4&0&x^{2}\\
0&5&0&3&0\\
1&0&8&0&x^{3}
\end{vmatrix}.
\]
\end{problem}

\begin{problem}{15 分}{9.5cm}
解齐次线性方程组
\[
\begin{cases}
x_1+x_3-x_4-3x_5=-2,\\
x_1+2x_2-x_3-x_5=1,\\
4x_1+6x_2-2x_3-4x_4+3x_5=7,\\
2x_1-2x_2+4x_3-7x_4+4x_5=1.
\end{cases}
\]
\end{problem}

\begin{problem}{15 分}{9cm}
\begin{enumerate}[label=(\arabic*),leftmargin=2.2em,itemsep=0.3em,topsep=0.3em]
  \item 试叙述矩阵秩的定义。
  \item 设 $\lambda$ 可取任意参数，试求矩阵
  $\begin{pmatrix}1&-1&1&1\\1&\lambda&1&1\\2&2\lambda&\lambda+4&3\end{pmatrix}$
  的秩。
\end{enumerate}
\end{problem}

\begin{problem}{15 分}{9.5cm}
试求解矩阵方程 $AXB=C$ 的解，其中
\[
A=\begin{pmatrix}1&2&3\\2&1&2\\1&3&4\end{pmatrix},\qquad
B=\begin{pmatrix}7&9\\4&5\end{pmatrix},\qquad
C=\begin{pmatrix}1&2\\1&0\\2&3\end{pmatrix}.
\]
\end{problem}

\begin{problem}{15 分}{9cm}
已知 $A'BA=2BA-12E$，其中
$A=\begin{pmatrix}2&0&0\\0&-2&1\\0&-1&1\end{pmatrix}$，求 $B$。
\end{problem}

\begin{problem}{15 分}{9cm}
试证明 $\begin{pmatrix}O&A\\B&C\end{pmatrix}$ 可逆并求其逆，这里 $A,B$ 分别为 $r$ 阶和 $s$ 阶可逆矩阵。
\end{problem}

\begin{problem}{15 分}{9.5cm}
\begin{enumerate}[label=(\arabic*),leftmargin=2.2em,itemsep=0.3em,topsep=0.3em]
  \item 试证明对矩阵 $A$ 实施一次初等行变换等价于用相应的初等矩阵左乘 $A$。
  \item 试证明方阵 $A$ 可逆的充分必要条件是 $|A|\ne 0$。
\end{enumerate}
\end{problem}
