\blockhead{填空题（85 分）}

\begin{problem}{}{4.5cm}
已知非齐次线性方程组
\[
\begin{cases}
x_1+x_2+x_3+x_4=-1,\\
4x_1+3x_2+5x_3-x_4=-1,\\
px_1+x_2+3x_3-qx_4=1
\end{cases}
\]
有解，且通解中有 2 个任意常数，则 $p=\blank$，$q=\blank$，方程组的系数矩阵的秩 $r=\blank$，通解为\blank[5.5cm]。
\end{problem}

\begin{problem}{}{5.5cm}
设
\[
A=\begin{pmatrix}1&1&1\\1&0&2\\4&4&a^2+3\end{pmatrix},\qquad
B=\begin{pmatrix}1&0&1\\1&3&3\\a+3&1-a&a^2+3\end{pmatrix}
\]
则当 $a=\blank$ 时，矩阵 $A$ 和 $B$ 不相抵（或不等价）；当 $a=\blank$ 时，矩阵 $A$ 和 $B$ 的秩都是 $2$；当矩阵 $A$ 和 $B$ 的秩都是 $3$ 时，$x_1=\blank$，$x_2=\blank$，$x_3=\blank$ 满足
$A\begin{pmatrix}x_1\\x_2\\x_3\end{pmatrix}=\begin{pmatrix}1\\3\\a^2+3\end{pmatrix}$。
\end{problem}

\blockhead{证明题（15 分）}

\begin{problem}{}{8.5cm}
设 $n$ 阶矩阵
$A=\begin{pmatrix}a_1\\a_2\\\vdots\\a_n\end{pmatrix}\begin{pmatrix}b_1&b_2&\cdots&b_n\end{pmatrix}$，
如果 $a_1b_1+a_2b_2+\cdots+a_nb_n=1$，证明 $A^{2}=A$。
\end{problem}

\begin{problem}{}{8.5cm}
设 $A,B,C$ 为 $n$ 阶方阵满足 $r(A)<r(C)\le n$，且 $BC=0$。证明存在非零的矩阵 $X$ 满足 $AX=BX$。
\end{problem}
