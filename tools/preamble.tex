% ---------- 题目环境 ----------
% \begin{problem}{15 分}{9cm} 题干 \end{problem}
% 第一参数：分值（留空则只显示题号）；第二参数：题后解答留白高度（解析模式下忽略）
\newcounter{problem}
\newcommand{\resetproblem}{\setcounter{problem}{0}}
\newcommand{\problemspace}{0pt}
\newenvironment{problem}[2]{%
  \par\addvspace{1.0em}%
  \ifsolutionsmode
    \needspace{4.5\baselineskip}%
  \else
    \needspace{\dimexpr#2+2.4\baselineskip\relax}%
  \fi
  \renewcommand{\problemspace}{#2}%
  \stepcounter{problem}%
  \noindent\textbf{\theproblem.}%
  \ifx\relax#1\relax\else\hspace{0.25em}\textbf{（#1）}\fi
  \hspace{0.4em}\ignorespaces
}{%
  \par\nobreak\ifsolutionsmode\vspace{0.45em}\else\vspace{\problemspace}\fi\par
}

% ---------- 解析环境（仅解析模式使用） ----------
% \begin{solution} 推导过程 \end{solution}：渲染为“解　推导过程”
\newenvironment{solution}{%
  \par\addvspace{0.15em}\nopagebreak[3]%
  \needspace{2.8\baselineskip}%
  \noindent\textbf{解}\hspace{0.6em}\ignorespaces
}{%
  \par\addvspace{0.55em}%
}

% \ans{...}：结论行，凡题目要求具体结果的都在解析末尾用它给出答案
\newcommand{\ans}[1]{%
  \par\addvspace{0.3em}\nopagebreak[3]%
  \noindent\textbf{答案}\hspace{0.6em}#1\par\addvspace{0.1em}%
}

% \rem{...}：评注，只在能点明关键思想或常见错处时用，每卷不超过三处
\newcommand{\rem}[1]{%
  \par\addvspace{0.3em}\nopagebreak[3]%
  \noindent{\small\textbf{注}\hspace{0.5em}#1}\par\addvspace{0.1em}%
}

% 填空作答横线：\blank 或 \blank[5cm]
\newcommand{\blank}[1][2.4cm]{\underline{\hspace{#1}}}

% 卷内分块标题，如 \blockhead{填空题（85 分）}
\newcommand{\blockhead}[1]{%
  \par\addvspace{1.3em}\nopagebreak[3]%
  \noindent{\large\bfseries #1}%
  \par\nopagebreak\addvspace{0.45em}%
}

% 每套试卷的卷头：\paperhead{学年}{学期}{考试}{目录条目}
\newcommand{\paperhead}[4]{%
  \clearpage
  \phantomsection
  \addcontentsline{toc}{section}{#4}%
  \renewcommand{\currentpaper}{#1\ #2}%
  \markboth{#4}{#4}%
  \begin{center}
    {\Large\bfseries\ziju{0.12} \schoolname}\\[0.45em]
    {\large \coursename}\\[0.6em]
    {\Large\bfseries #1 学年\quad #2\quad #3考试\ifsolutionsmode 试题解析\else 试卷\fi}\\[0.65em]
    \rule{\linewidth}{1.1pt}
  \end{center}
  \vspace{0.4em}
  \resetproblem
}
